# 상수 컬럼 제거의 필요성 및 방법론

## 개요
산탄데르 고객 만족 예측 데이터셋에서 발견된 **34개의 상수 컬럼(모든 값이 0)**은 반드시 제거되어야 합니다. 이는 모델 성능 향상, 계산 효율성, 그리고 통계적 안정성을 위한 필수 전처리 단계입니다.

---

## 1. 상수 컬럼이란?

### 정의
- **상수 컬럼**: 모든 행에서 동일한 값을 가지는 변수
- **준상수 컬럼**: 95% 이상의 행에서 동일한 값을 가지는 변수

### 산탄데르 데이터의 현황
```python
# 상수 변수 현황
- 완전 상수 (분산 = 0): 34개 컬럼
- 준상수 (95% 이상 동일): 약 100개 추가 컬럼
- 전체 변수 대비: 34/371 = 9.2%
```

## 2. 제거가 필요한 이유

### 2.1 통계학적 관점

#### 분산의 중요성
```python
# 분산이 0인 변수의 문제점
# 1. 정보량 없음
# 2. 수치적 불안정성 초래
# 3. 통계적 계산 오류 발생
Variance(X) = 0 → Standard Deviation(X) = 0

# 표준화 불가능
# 1. Z-점수 계산 불가
# 2. 상관계수 계산 불가
# 3. 회귀 계수 추정 불가
# 4. PCA 등 차원 축소 불가
Z-score = (X - mean) / std  # std = 0이면 division by zero error

# 상관계수 계산 불가
# 1. 두 변수 간 관계 파악 불가
# 2. 다중공선성 문제 발생 가능
# 3. 피어슨 상관계수 정의 불가
# 4. 회귀 분석에서 영향력 없음
Correlation = Cov(X,Y) / (std(X) * std(Y))  # std(X) = 0이면 undefined
```

#### 정보 이론 관점
```python
# 엔트로피 = 0 (정보량 없음)
Entropy(X) = -Σ p(xi) * log(p(xi))
# 모든 값이 동일하면 p(x) = 1, Entropy = 0

# 정보 이득 = 0 (예측력 없음)
Information_Gain = Entropy(before) - Entropy(after) = 0
```

### 2.2 머신러닝 관점

#### 모델별 영향
```python
# 1. 선형 모델 (Logistic Regression, SVM)
- 특이행렬(Singular Matrix) 문제 발생
- 계수 추정 불가능
- 수렴 실패

# 2. 트리 기반 모델 (Random Forest, XGBoost)
- 분할 기준으로 사용 불가 (정보 이득 = 0)
- 불필요한 메모리 사용
- Feature Importance = 0

# 3. 신경망 (Neural Networks)
- 그래디언트 = 0 (학습 불가)
- 가중치 업데이트 없음
- Dead Neurons 발생

# 4. 거리 기반 모델 (KNN, K-Means)
- 거리 계산에 영향 없음
- 차원만 증가시켜 계산 비용 증가
```

#### 차원의 저주 (Curse of Dimensionality)
```python
# 불필요한 차원 증가의 문제
- 데이터 희소성 증가
- 거리 척도의 의미 상실
- 과적합 위험 증가
- 계산 복잡도 증가: O(n*d) → O(n*(d-34))
```

### 2.3 실무적 관점

#### 계산 효율성
```python
# 메모리 사용량 감소
memory_saved = n_samples * n_constant_features * bytes_per_float

# 학습 시간 단축
- 특성 수 감소: 371 → 337 (9.2% 감소)

```

#### 모델 해석성

```python
- 노이즈 제거로 진짜 중요한 변수 식별 용이
```

## 3. 제거 방법론

### 3.1 상수 변수 식별

```python
import pandas as pd
import numpy as np
from sklearn.feature_selection import VarianceThreshold

def identify_constant_columns(df):
    """상수 컬럼 식별"""

    # 방법 1: 표준편차 사용
    constant_cols = df.columns[df.std() == 0].tolist()

    # 방법 2: 고유값 개수 사용
    constant_cols = [col for col in df.columns if df[col].nunique() == 1]

    # 방법 3: VarianceThreshold 사용
    selector = VarianceThreshold(threshold=0)
    selector.fit(df)
    constant_cols = df.columns[~selector.get_support()].tolist()

    return constant_cols

# 실행
```

### 3.2 준상수 변수 식별

```python
def identify_quasi_constant(df, threshold=0.95):
    """준상수 컬럼 식별 (95% 이상 동일값)"""

    quasi_constant = []
    for col in df.columns:
        # 최빈값의 비율 계산
        dominant_ratio = df[col].value_counts(normalize=True).iloc[0]
        if dominant_ratio > threshold:
            quasi_constant.append(col)

    return quasi_constant

# 실행

print(f"준상수 컬럼 개수: {len(quasi_constant_cols)}")
```

### 3.3 안전한 제거 프로세스

```python
def safe_remove_constant_features(X_train, X_test, X_val=None):
    """훈련/검증/테스트 세트에서 일관되게 상수 변수 제거"""

    # 1. 훈련 세트 기준으로 상수 변수 식별
    selector = VarianceThreshold(threshold=0)
    selector.fit(X_train)

    # 2. 모든 세트에 동일하게 적용
    X_train_selected = selector.transform(X_train)
    X_test_selected = selector.transform(X_test)

    if X_val is not None:
        X_val_selected = selector.transform(X_val)
    else:
        X_val_selected = None

    # 3. 제거된 컬럼 정보 저장
    removed_features = X_train.columns[~selector.get_support()].tolist()
    selected_features = X_train.columns[selector.get_support()].tolist()

    # 4. 로깅
    print(f"원본 특성 수: {X_train.shape[1]}")
    print(f"제거된 특성 수: {len(removed_features)}")
    print(f"남은 특성 수: {len(selected_features)}")

    return (X_train_selected, X_test_selected, X_val_selected,
            selected_features, removed_features)
```

## 4. 실제 적용 예시

### 4.1 전체 전처리 파이프라인

```python
class SantanderPreprocessor:
    """산탄데르 데이터 전처리 클래스"""

    def __init__(self, variance_threshold=0, correlation_threshold=0.98):
        self.variance_threshold = variance_threshold
        self.correlation_threshold = correlation_threshold
        self.constant_features = []
        self.quasi_constant_features = []
        self.duplicate_features = []
        self.selected_features = []

    def fit(self, X, y=None):
        """전처리 파라미터 학습"""

        # 1. 상수 변수 식별
        self.constant_selector = VarianceThreshold(self.variance_threshold)
        self.constant_selector.fit(X)
        self.constant_features = X.columns[~self.constant_selector.get_support()].tolist()

        # 2. 준상수 변수 식별 (남은 변수 중에서)
        X_after_constant = X[X.columns[self.constant_selector.get_support()]]
        self.quasi_constant_features = self._find_quasi_constant(X_after_constant)

        # 3. 중복 변수 식별
        remaining_cols = [col for col in X_after_constant.columns
                         if col not in self.quasi_constant_features]
        self.duplicate_features = self._find_duplicates(X[remaining_cols])

        # 4. 최종 선택된 특성
        self.selected_features = [col for col in remaining_cols
                                 if col not in self.duplicate_features]

        return self

    def transform(self, X):
        """전처리 적용"""
        return X[self.selected_features]

    def fit_transform(self, X, y=None):
        """학습 및 변환"""
        return self.fit(X, y).transform(X)

    def _find_quasi_constant(self, X, threshold=0.95):
        """준상수 변수 찾기"""
        quasi_constant = []
        for col in X.columns:
            if X[col].value_counts(normalize=True).iloc[0] > threshold:
                quasi_constant.append(col)
        return quasi_constant

    def _find_duplicates(self, X, threshold=0.98):
        """중복 변수 찾기"""
        corr_matrix = X.corr().abs()
        upper_triangle = corr_matrix.where(
            np.triu(np.ones(corr_matrix.shape), k=1).astype(bool)
        )
        duplicates = [column for column in upper_triangle.columns
                     if any(upper_triangle[column] > threshold)]
        return duplicates

    def get_preprocessing_report(self):
        """전처리 결과 리포트"""
        report = f"""
        ========== 전처리 결과 리포트 ==========

        1. 상수 변수 (분산 = 0)
           - 제거된 개수: {len(self.constant_features)}
           - 변수명: {self.constant_features[:5]}...

        2. 준상수 변수 (95% 이상 동일)
           - 제거된 개수: {len(self.quasi_constant_features)}

        3. 중복 변수 (상관계수 > 0.98)
           - 제거된 개수: {len(self.duplicate_features)}

        4. 최종 결과
           - 최종 선택된 변수: {len(self.selected_features)}개
           - 제거된 총 변수: {len(self.constant_features) +
                          len(self.quasi_constant_features) +
                          len(self.duplicate_features)}개

        =====================================
        """
        return report

# 사용 예시
preprocessor = SantanderPreprocessor(variance_threshold=0)
X_train_processed = preprocessor.fit_transform(X_train)
X_test_processed = preprocessor.transform(X_test)
print(preprocessor.get_preprocessing_report())
```

### 4.2 성능 비교 실험

```python
import time
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from xgboost import XGBClassifier
from sklearn.metrics import roc_auc_score

def compare_performance(X_train, y_train, X_val, y_val):
    """상수 변수 제거 전후 성능 비교"""

    results = {}

    # 모델 리스트
    models = {
        'Logistic': LogisticRegression(max_iter=1000),
        'RandomForest': RandomForestClassifier(n_estimators=100),
        'XGBoost': XGBClassifier(n_estimators=100)
    }

    for name, model in models.items():
        # 원본 데이터로 학습
        start = time.time()
        model.fit(X_train, y_train)
        train_time = time.time() - start
        pred = model.predict_proba(X_val)[:, 1]
        auc = roc_auc_score(y_val, pred)

        results[name] = {
            'train_time': train_time,
            'auc': auc,
            'n_features': X_train.shape[1]
        }

    return results

# 실험 실행
print("===== 상수 변수 포함 =====")
results_with_constant = compare_performance(X_train, y_train, X_val, y_val)

print("\n===== 상수 변수 제거 =====")
X_train_clean = remove_constant_features(X_train)
X_val_clean = remove_constant_features(X_val)
results_without_constant = compare_performance(X_train_clean, y_train,
                                              X_val_clean, y_val)

# 결과 비교
for model in results_with_constant.keys():
    before = results_with_constant[model]
    after = results_without_constant[model]

    print(f"\n{model}:")
    print(f"  AUC: {before['auc']:.4f} → {after['auc']:.4f} "
          f"({(after['auc']-before['auc']):.4f} 개선)")
    print(f"  학습시간: {before['train_time']:.2f}s → {after['train_time']:.2f}s "
          f"({(1-after['train_time']/before['train_time'])*100:.1f}% 단축)")
```
